document.addEventListener('DOMContentLoaded', async function() {
    // First check authentication

    const tiles = document.querySelectorAll('.tile');

    function startLoading() {
        if (predictButton) {
            predictButton.classList.add('loading');
            const content = predictButton.querySelector('.content');
            const loader = predictButton.querySelector('.loader');
            if (content) content.style.visibility = 'hidden';
            if (loader) loader.style.display = 'flex';
        }
    }

    function stopLoading() {
        if (predictButton) {
            predictButton.classList.remove('loading');
            const content = predictButton.querySelector('.content');
            const loader = predictButton.querySelector('.loader');
            if (content) content.style.visibility = 'visible';
            if (loader) loader.style.display = 'none';
        }
    }

    // Function to check if inputs are valid
    async function validateServerSeed(seed) {
        if (!seed || seed.length !== 64) {
            showError('Server seed must be 64 characters long');
            return false;
        }
        
        try {
            const response = await fetch('https://diceapi.pythonanywhere.com/validate_seed', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({ server_seed: seed })
            });
            
            const data = await response.json();
            
            // Handle both successful and error responses
            if (response.ok && data.valid) {
                clearError();
                return true;
            } else {
                showError(data.error || 'Invalid server seed');
                return false;
            }
        } catch (error) {
            console.error('Validation error:', error);
            showError('Failed to validate server seed');
            return false;
        }
    }

    // Function to check if inputs are valid
    async function checkInputs() {
        const serverSeed = serverSeedInput?.value?.trim() || '';
        const betAmount = betAmountInput?.value?.trim() || '';
        const mines = minesSelect?.value || '';

        // Add console.log for debugging
        console.log('Checking inputs...');

        const isServerSeedValid = await validateServerSeed(serverSeed);
        const isBetAmountValid = betAmount !== '' && !isNaN(betAmount) && parseFloat(betAmount) > 0;
        const isMinesValid = mines !== '' && !isNaN(mines) && parseInt(mines) >= 1;

        // Add console.log for debugging
        console.log('Validation results:', {
            isServerSeedValid,
            isBetAmountValid,
            isMinesValid
        });

        if (predictButton) {
            const shouldEnable = isServerSeedValid && isBetAmountValid && isMinesValid;
            predictButton.disabled = !shouldEnable;
            predictButton.style.opacity = shouldEnable ? '1' : '0.5';
            predictButton.style.cursor = shouldEnable ? 'pointer' : 'default';
            
            // Add console.log for debugging
            console.log('Button state:', {
                disabled: predictButton.disabled,
                opacity: predictButton.style.opacity,
                cursor: predictButton.style.cursor
            });
        }
    }

    // Add event listeners to all inputs

        betAmountInput.addEventListener('input', checkInputs);

    // Generate random positions for gems with adjusted count based on mines
    function generateGemPositions(minesCount) {
        let gemCount;
        switch(minesCount) {
            case 1: gemCount = Math.floor(Math.random() * 4) + 8; // 8-10 gems
            break;
            case 2: gemCount = Math.floor(Math.random() * 3) + 3; // 3-5 gems
            break;
            case 3: gemCount = Math.floor(Math.random() * 4) + 2; // 2-5 gems
            break;
            case 4: gemCount = Math.floor(Math.random() * 2) + 2; // 2-3 gems
            break;
            case 5: gemCount = Math.floor(Math.random() * 4) + 1; // 1-3 gems
            break;
            case 6: gemCount = Math.floor(Math.random() * 3) + 1; // 1-2 gems
            break;
            case 7:
            case 8: gemCount = 1; // 1 gem
            break;
            default: gemCount = 1;
        }

        const positions = new Set();
        while (positions.size < gemCount) {
            positions.add(Math.floor(Math.random() * 25));

                        }
                        
                        // Clear game content
                        while (gameContent.firstChild) {
                            gameContent.removeChild(gameContent.firstChild);
                        }
                        
                        // Add new grid
                        gameContent.appendChild(newGrid);
                        
                        // Fade in new grid
                        newGrid.style.opacity = '0';
                        requestAnimationFrame(() => {
                            newGrid.style.opacity = '1';
                            newGrid.style.transition = 'opacity 0.3s ease';
                        });

                        // Reveal patterns with delay
                        setTimeout(() => {
                            revealPattern(data.gems, 'gem', newGrid);
                            revealPattern(data.bombs, 'bomb', newGrid);
                        }, 300);
                    } else {
                        throw new Error(data.error || 'Failed to generate pattern');
                    }
                } catch (error) {
                    console.error('Error:', error);
                    showError(error.message);
                } finally {
                    stopLoading();
                }
            }
        });
    }

    // Initialize on page load
    initializeGrid();
    checkInputs(); // Check initial state of inputs
});
